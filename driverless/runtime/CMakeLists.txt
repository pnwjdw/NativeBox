cmake_minimum_required(VERSION 3.25)
project(NativeBoxInstrumentedRuntime VERSION 0.4.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "NativeBox runtime requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(NATIVEBOX_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${NATIVEBOX_SOURCE_ROOT}/scripts/msvc-dependencies.cmake")
if(NOT EXISTS "${NATIVEBOX_DETOURS_ROOT}/src/detours.h")
  message(FATAL_ERROR "Pinned official Detours sources are required")
endif()
set(DETOURS_SOURCES detours.cpp modules.cpp disasm.cpp image.cpp creatwth.cpp disolx86.cpp disolx64.cpp disolia64.cpp disolarm.cpp disolarm64.cpp)
list(TRANSFORM DETOURS_SOURCES PREPEND "${NATIVEBOX_DETOURS_ROOT}/src/")
add_library(nativebox_detours STATIC ${DETOURS_SOURCES})
set_target_properties(nativebox_detours PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON)
target_include_directories(nativebox_detours PUBLIC "${NATIVEBOX_DETOURS_ROOT}/src")
target_compile_definitions(nativebox_detours PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(nativebox_detours PRIVATE /W4 /Gy /utf-8)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(NATIVEBOX_RUNTIME_BITS 64)
else()
  set(NATIVEBOX_RUNTIME_BITS 32)
endif()
set(NATIVEBOX_RUNTIME_NAME "nativebox_runtime${NATIVEBOX_RUNTIME_BITS}")
configure_file(runtime.rc.in "${CMAKE_CURRENT_BINARY_DIR}/runtime.rc" @ONLY)
add_library(${NATIVEBOX_RUNTIME_NAME} SHARED base.cpp files.cpp registry.cpp network.cpp bootstrap.cpp hooks.cpp telemetry.cpp "runtime${NATIVEBOX_RUNTIME_BITS}.def" "${CMAKE_CURRENT_BINARY_DIR}/runtime.rc")
target_include_directories(${NATIVEBOX_RUNTIME_NAME} PRIVATE "${NATIVEBOX_SOURCE_ROOT}")
target_compile_definitions(${NATIVEBOX_RUNTIME_NAME} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NB_OFFICIAL_DETOURS)
target_compile_options(${NATIVEBOX_RUNTIME_NAME} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>")
target_link_libraries(${NATIVEBOX_RUNTIME_NAME} PRIVATE nativebox_detours advapi32 bcrypt ws2_32)
target_compile_options(${NATIVEBOX_RUNTIME_NAME} PRIVATE /Zi)
target_link_options(${NATIVEBOX_RUNTIME_NAME} PRIVATE /DEBUG /OPT:REF /OPT:ICF)
install(TARGETS ${NATIVEBOX_RUNTIME_NAME} RUNTIME DESTINATION .)
